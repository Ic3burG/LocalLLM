# MLX Stable Diffusion — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add on-device image generation via MLX Stable Diffusion, accessible via an image mode toggle in the chat UI and as an agent tool.

**Architecture:** New `image_pipeline.py` owns the SD lifecycle and model swap logic. Two new FastAPI routes in `gemma_bridge.py` expose generation and model discovery. A proxy entry in `server.js` bridges the frontend. A new async tool in `agent_utils.py` lets the LLM generate images conversationally, with agent.py emitting a dedicated SSE event when an image is produced. Frontend additions in `index.html` provide a mode pill toggle, inline controls, shimmer loading card, image action bar, and lightbox. All image data flows through `data-*` attributes and module-level state — no inline event handlers with embedded JSON.

**Tech Stack:** Python (MLX, mlx-stable-diffusion, Pillow), FastAPI, Express.js proxy, vanilla JS / Tailwind CSS (frontend), pytest + unittest.mock (tests)

  ```python
  import io
  import base64
  import logging
  import threading
  import time
  from pathlib import Path

  logger = logging.getLogger("image_pipeline")

  MLX_MODELS_DIR = Path(__file__).parent / "mlx_models"
  SD_MODEL_DIR = MLX_MODELS_DIR / "sd-1.5"

  STYLE_SUFFIXES: dict[str, str] = {
      "default": "",
      "photorealistic": ", photorealistic, 8k, highly detailed, sharp focus",
      "anime": ", anime style, cel shaded, vibrant colors",
      "sketch": ", pencil sketch, black and white, hand drawn",
  }

  _FAST_MODELS = {"phi-4-mini-4bit", "deepseek-v4-mini-7b-4bit"}

  _sd_model = None
  _sd_lock = threading.Lock()


  def _apply_style(prompt: str, style: str) -> str:
      return prompt + STYLE_SUFFIXES.get(style, "")


  def _parse_size(size: str) -> tuple[int, int]:
      w, h = size.lower().split("x")
      return int(w), int(h)


  def _load_sd_model(model_path: str):
      from mlx_stable_diffusion import StableDiffusion
      return StableDiffusion(model_path)


  def _run_diffusion(model, prompt: str, width: int, height: int, steps: int) -> bytes:
      """Run the SD diffusion loop. Returns raw PNG bytes."""
      import mlx.core as mx
      from PIL import Image
      import numpy as np

      latents = model.generate_latents(
          prompt,
          n_images=1,
          num_steps=steps,
          cfg_weight=7.5,
          image_size=(height, width),
      )
      x_t = None
      for x_t in latents:
          mx.eval(x_t)
      pil_image = model.decode(x_t)[0]
      buf = io.BytesIO()
      pil_image.save(buf, format="PNG")
      return buf.getvalue()


  def _should_swap(loaded_models: list[str]) -> bool:
      """Return True if we need to unload the text model before loading SD."""
      if not loaded_models:
          return False
      return not any(m in _FAST_MODELS for m in loaded_models)


  def _unload_text_model() -> None:
      """Clear MLX metal cache to free VRAM for SD."""
      try:
          import mlx.core as mx
          mx.metal.clear_cache()
          logger.info("text model cache cleared for SD swap")
      except Exception as e:
          logger.warning("metal cache clear failed: %s", e)


  def generate_image(
      prompt: str,
      size: str = "512x512",
      steps: int = 20,
      style: str = "default",
  ) -> dict:
      """Generate an image. Returns dict with image_b64 and metadata."""
      raise NotImplementedError("implement in Task 4")


  def get_downloaded_models() -> list[str]:
      """Return list of downloaded SD model IDs."""
      models = []
      if SD_MODEL_DIR.exists():
          models.append("sd-1.5")
      return models
  ```

  ```python
  def generate_image(
      prompt: str,
      size: str = "512x512",
      steps: int = 20,
      style: str = "default",
  ) -> dict:
      """Generate an image. Returns dict with image_b64 and metadata."""
      global _sd_model

      if not SD_MODEL_DIR.exists():
          raise FileNotFoundError("model_not_found")

      width, height = _parse_size(size)
      styled_prompt = _apply_style(prompt, style)

      with _sd_lock:
          loaded = get_loaded_models()
          if _should_swap(loaded):
              _unload_text_model()

          if _sd_model is None:
              logger.info("loading SD model from %s", SD_MODEL_DIR)
              _sd_model = _load_sd_model(str(SD_MODEL_DIR))

          t0 = time.monotonic()
          png_bytes = _run_diffusion(_sd_model, styled_prompt, width, height, steps)
          elapsed_ms = int((time.monotonic() - t0) * 1000)

      image_b64 = base64.b64encode(png_bytes).decode("utf-8")
      return {
          "image_b64": image_b64,
          "width": width,
          "height": height,
          "steps": steps,
          "elapsed_ms": elapsed_ms,
      }
  ```

  ```python
  @app.post("/v1/image/generate")
  async def generate_image_route(request: Request):
      body = await request.json()
      prompt = body.get("prompt", "").strip()
      if not prompt:
          return JSONResponse({"error": "prompt_required"}, status_code=400)
      size = body.get("size", "512x512")
      steps = int(body.get("steps", 20))
      style = body.get("style", "default")

      loop = asyncio.get_event_loop()
      try:
          result = await asyncio.wait_for(
              loop.run_in_executor(
                  None,
                  lambda: image_pipeline.generate_image(prompt, size, steps, style),
              ),
              timeout=120.0,
          )
          return JSONResponse(result)
      except FileNotFoundError:
          return JSONResponse({"error": "model_not_found"}, status_code=503)
      except asyncio.TimeoutError:
          return JSONResponse({"error": "timeout"}, status_code=504)
      except MemoryError:
          return JSONResponse({"error": "insufficient_memory"}, status_code=507)
      except Exception as e:
          logger.error("image generation failed: %s", e, exc_info=True)
          return JSONResponse({"error": "generation_failed", "detail": str(e)}, status_code=500)


  @app.get("/v1/image/models")
  async def image_models_route():
      downloaded = image_pipeline.get_downloaded_models()
      return JSONResponse({"available": ["sd-1.5"], "downloaded": downloaded})
  ```

  ```python
  async def _generate_image(prompt: str, size: str = "512x512", steps: int = 20) -> str:
      log_audit(f"GENERATE_IMAGE: prompt={prompt!r} size={size} steps={steps}")
      try:
          import requests as _requests
          loop = asyncio.get_event_loop()
          payload = {"prompt": prompt, "size": size, "steps": int(steps)}

          def _sync_post():
              return _requests.post(
                  "http://localhost:9379/v1/image/generate",
                  json=payload,
                  timeout=125,
              )

          resp = await loop.run_in_executor(None, _sync_post)
          data = resp.json()

          if resp.status_code != 200 or "error" in data:
              error = data.get("error", "generation_failed")
              return f"ERROR: Image generation failed — {error}"

          return json.dumps({
              "__image__": True,
              "image_b64": data["image_b64"],
              "width": data["width"],
              "height": data["height"],
              "steps": data["steps"],
              "elapsed_ms": data["elapsed_ms"],
              "prompt": prompt,
              "size": size,
          })
      except Exception as e:
          logger.error("_generate_image failed: %s", e)
          return f"ERROR: {e}"
  ```

  ```javascript
  async function submitImageGeneration(prompt, size, steps, style) {
    if (!prompt) return;
    appendMessage("user", prompt);
    let isWarmup = false;
    try {
      const modelsRes = await fetch("/api/image/models");
      const modelsData = await modelsRes.json();
      isWarmup = modelsData.downloaded.length === 0;
    } catch (_) {}
    if (isWarmup) {
      document.getElementById("warmup-banner").classList.add("visible");
    }
    const shimmer = appendImageShimmer(isWarmup);
    try {
      const res = await fetch("/api/image/generate", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          prompt,
          size,
          steps: parseInt(steps, 10),
          style,
        }),
      });
      const data = await res.json();
      shimmer.remove();
      document.getElementById("warmup-banner").classList.remove("visible");
      if (!res.ok || data.error) {
        const errorMessages = {
          model_not_found:
            "Image model not downloaded. Run scripts/download_sd.sh to install it.",
          insufficient_memory:
            "Not enough memory. Try a smaller size or switch to a lighter text model.",
          timeout: "Generation timed out. Try fewer steps or a smaller size.",
        };
        appendMessage(
          "assistant",
          errorMessages[data.error] || "Image generation failed: " + data.error
        );
        return;
      }
      appendImageCard({ ...data, prompt, size, style });
    } catch (err) {
      shimmer.remove();
      appendMessage("assistant", "Image generation error: " + err.message);
    }
  }
  ```
